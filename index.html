// Configured Web App Webhook URL const GOOGLE_SCRIPT_URL = "YOUR_NEW_DEPLOYED_WEB_APP_URL_HERE"; // Auto-fetch real-time logs from sheet every 5 seconds for Multi-Device Syncing async function syncLogsFromSheet() { if(!GOOGLE_SCRIPT_URL) return; try { let res = await fetch(GOOGLE_SCRIPT_URL + "?action=getLogs"); let json = await res.json(); if(json.status === "success" && json.data) { renderLogsTable(json.data); } } catch(e) { console.error("Auto-sync error", e); } } setInterval(syncLogsFromSheet, 5000);
