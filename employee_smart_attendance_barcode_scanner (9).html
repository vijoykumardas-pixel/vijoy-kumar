Google Apps Script Code (Sheet ID: 117_aY6...)

Step-by-Step Setup Guide:

  1. Open your Google Sheet: Open Sheet
  2. Go to top menu: Extensions > Apps Script.
  3. Replace everything in `Code.gs` with the snippet below and click Save (Ctrl + S).
  4. Click Deploy > New deployment (top right).
  5. Select type: Web app.
  6. Set "Execute as" = Me, and "Who has access" = Anyone.
  7. Click Deploy, copy the generated Web App URL, and paste it into the Shift & Config settings of this app!

function getTargetSpreadsheet() {
  var SHEET_ID = "117_aY6-I5Fl26qMnyfE7H1d9vRGyp4an3vbr0hewKXg";
  try {
    return SpreadsheetApp.openById(SHEET_ID);
  } catch(e) {
    return SpreadsheetApp.getActiveSpreadsheet();
  }
}

function getLogSheet() {
  var ss = getTargetSpreadsheet();
  var logSheetName = "Attendance Logs";
  var logSheet = ss.getSheetByName(logSheetName);

  if (!logSheet) {
    logSheet = ss.insertSheet(logSheetName);
    logSheet.appendRow([
      "ID", "Timestamp", "Date", "Time", "Employee ID / HRMS",
      "Workforce Code", "Employee Name", "Department", "Designation",
      "Action", "Late Status", "Late Minutes", "Break Duration (Mins)", "Raw Data Dump"
    ]);
    logSheet.getRange("A1:N1").setFontWeight("bold").setBackground("#1e293b").setFontColor("#ffffff");
    logSheet.setFrozenRows(1);
  }
  return logSheet;
}

function doPost(e) {
  return handleIncomingLog(e);
}

function doGet(e) {
  // Support appending via GET query param (bulletproof fallback for browser CORS)
  if (e && e.parameter && e.parameter.payload) {
    return handleIncomingLog(e);
  }

  try {
    var logSheet = getLogSheet();
    var data = logSheet.getDataRange().getValues();

    if (data.length <= 1) {
      return ContentService.createTextOutput(JSON.stringify({ logs: [] }))
        .setMimeType(ContentService.MimeType.JSON);
    }

    var logs = [];
    for (var i = 1; i < data.length; i++) {
      var row = data[i];
      if (!row[0] && !row[4]) continue;

      logs.push({
        id: String(row[0] || ("LOG-" + i)),
        timestamp: row[1] ? new Date(row[1]).toISOString() : new Date().toISOString(),
        date: String(row[2] || ""),
        time: String(row[3] || ""),
        hrmsCode: String(row[4] || ""),
        workforceCode: String(row[5] || ""),
        name: String(row[6] || ""),
        dept: String(row[7] || ""),
        designation: String(row[8] || ""),
        action: String(row[9] || "IN"),
        isLate: String(row[10]) === "LATE",
        lateMinutes: Number(row[11]) || 0,
        breakDurationMinutes: Number(row[12]) || 0,
        syncStatus: "Synced"
      });
    }

    // Newest first for UI display
    logs.reverse();

    return ContentService.createTextOutput(JSON.stringify({ logs: logs }))
      .setMimeType(ContentService.MimeType.JSON);

  } catch(err) {
    return ContentService.createTextOutput(JSON.stringify({ error: err.toString(), logs: [] }))
      .setMimeType(ContentService.MimeType.JSON);
  }
}

function handleIncomingLog(e) {
  try {
    var raw = "";
    if (e && e.postData && e.postData.contents) {
      raw = e.postData.contents;
    } else if (e && e.parameter && e.parameter.payload) {
      raw = e.parameter.payload;
    }

    if (!raw) {
      return ContentService.createTextOutput(JSON.stringify({ status: "error", message: "No data" }))
        .setMimeType(ContentService.MimeType.JSON);
    }

    var data = typeof raw === "string" ? JSON.parse(raw) : raw;
    var logSheet = getLogSheet();

    if (data.action === "CLEAR_ALL_LOGS") {
      var lastRow = logSheet.getLastRow();
      if (lastRow > 1) {
        logSheet.deleteRows(2, lastRow - 1);
      }
      return ContentService.createTextOutput(JSON.stringify({ status: "success", message: "Cleared" }))
        .setMimeType(ContentService.MimeType.JSON);
    }

    var now = new Date();
    logSheet.appendRow([
      data.id || ("LOG-" + now.getTime()),
      now,
      data.date || Utilities.formatDate(now, Session.getScriptTimeZone(), "yyyy-MM-dd"),
      data.time || Utilities.formatDate(now, Session.getScriptTimeZone(), "hh:mm:ss a"),
      data.hrmsCode || "",
      data.workforceCode || "",
      data.name || "Unknown",
      data.dept || "",
      data.designation || "",
      data.action || "IN",
      data.isLate ? "LATE" : "ON_TIME",
      data.lateMinutes || 0,
      data.breakDurationMinutes || 0,
      JSON.stringify(data)
    ]);

    return ContentService.createTextOutput(JSON.stringify({ status: "success" }))
      .setMimeType(ContentService.MimeType.JSON);

  } catch(error) {
    return ContentService.createTextOutput(JSON.stringify({ status: "error", message: error.toString() }))
      .setMimeType(ContentService.MimeType.JSON);
  }
}
