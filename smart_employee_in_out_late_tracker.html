Google Apps Script Backend Code (`Code.gs`)

Copy this code into your Google Sheet's Apps Script Editor to enable live database sync.

Instructions to Deploy:

  1. Open your Google Sheet → Go to Extensions → Apps Script.
  2. Paste the code below into Code.gs replacing existing text.
  3. Click Deploy → New deployment → Select type: Web App.
  4. Set Execute as: Me and Who has access: Anyone.
  5. Copy the Web App URL and paste it in the Sheet Config button of this app.

// SMART EMPLOYEE TRACKER - GOOGLE APPS SCRIPT
// ==========================================

function doGet(e) {
  var action = e.parameter.action;
  var sheet = getOrCreateLogsSheet();
  
  if (action === "fetchLogs") {
    var data = sheet.getDataRange().getValues();
    var headers = data.shift();
    var logs = data.map(function(row) {
      return {
        id: row[0],
        timestamp: row[1],
        empId: row[2],
        empName: row[3],
        dept: row[4],
        action: row[5],
        lateCalc: row[6],
        breakDuration: row[7],
        status: row[8]
      };
    });
    return responseJSON({ status: "success", logs: logs });
  }

  return responseJSON({ status: "online", message: "Employee Tracker Webhook Active" });
}

function doPost(e) {
  try {
    var contents = JSON.parse(e.postData.contents);
    var sheet = getOrCreateLogsSheet();
    
    sheet.appendRow([
      contents.id || Date.now(),
      contents.timestamp || new Date().toLocaleString(),
      contents.empId || "",
      contents.empName || "",
      contents.dept || "",
      contents.action || "",
      contents.lateCalc || "-",
      contents.breakDuration || "-",
      "Synced"
    ]);

    return responseJSON({ status: "success", message: "Row added successfully" });
  } catch (err) {
    return responseJSON({ status: "error", message: err.toString() });
  }
}

function getOrCreateLogsSheet() {
  var ss = SpreadsheetApp.getActiveSpreadsheet();
  var sheet = ss.getSheetByName("Attendance Logs");
  if (!sheet) {
    sheet = ss.insertSheet("Attendance Logs");
    sheet.appendRow([
      "ID", "Scan Time & Date", "EMP ID", "Employee Name", "Department & Role", "Action", "Late Calculation", "Break Duration", "Status"
    ]);
    sheet.getRange(1, 1, 1, 9).setFontWeight("bold").setBackground("#1e293b").setFontColor("#ffffff");
  }
  return sheet;
}

function responseJSON(obj) {
  return ContentService.createTextOutput(JSON.stringify(obj))
    .setMimeType(ContentService.MimeType.JSON);
}
